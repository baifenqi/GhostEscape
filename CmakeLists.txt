# 设置CMake最低版本要求
cmake_minimum_required(VERSION 3.15)

# 定义项目名称、版本和使用的编程语言
project(GHOSTESCAPE VERSION 1.0.0 LANGUAGES C CXX)

# 设置C++标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 设置 UTF-8 编码支持（必须在 add_executable 之前）
if(MSVC)
    add_compile_options(/utf-8)
endif()

# 设置输出目录
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")

# 支持多配置生成器（如 Visual Studio）
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Choose the type of build." FORCE)
endif()

# 查找所需的 SDL3 相关库
find_package(SDL3 REQUIRED)
find_package(SDL3_image REQUIRED)
find_package(SDL3_mixer REQUIRED)
find_package(SDL3_ttf REQUIRED)
find_package(glm REQUIRED)

# 添加可执行文件及其源文件
add_executable(GHOSTESCAPE
    src/main.cpp
    src/core/Game.cpp
    src/core/Scene.cpp
    src/core/Object.cpp
    src/core/ObjectWorld.cpp
    src/core/ObjectAffiliate.cpp
    src/core/Actor.cpp
    src/core/Asset_store.cpp

    src/affiliate/Sprite.cpp
    src/affiliate/SpriteAnim.cpp
    src/affiliate/Collider.cpp

    src/SceneMain.cpp
    src/Player.cpp
    src/Enemy.cpp
)

# 根据编译器类型设置编译选项
# 注意：UTF-8 已经通过 add_compile_options(/utf-8) 全局设置，此处不再重复
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(GHOSTESCAPE PRIVATE
        $<$<CONFIG:Debug>:/W4 /Od /Zi>
        $<$<CONFIG:Release>:/O2 /DNDEBUG>
    )
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(GHOSTESCAPE PRIVATE
        $<$<CONFIG:Debug>:-g -O0 -Wall -Wextra>
        $<$<CONFIG:Release>:-O3 -DNDEBUG>
    )
endif()

# 设置包含目录
target_include_directories(GHOSTESCAPE PUBLIC
    ${PROJECT_SOURCE_DIR}/include
    ${SDL3_INCLUDE_DIRS}
    ${SDL3_IMAGE_INCLUDE_DIRS}
    ${SDL3_MIXER_INCLUDE_DIRS}
    ${SDL3_TTF_INCLUDE_DIRS}
)

# 链接目标所需的库
target_link_libraries(GHOSTESCAPE PRIVATE
    SDL3::SDL3
    SDL3_image::SDL3_image
    SDL3_mixer::SDL3_mixer
    SDL3_ttf::SDL3_ttf
    glm::glm
)

# 跨平台适配
if(WIN32)
    target_compile_definitions(GHOSTESCAPE PRIVATE PLATFORM_WINDOWS)
elseif(APPLE)
    target_compile_definitions(GHOSTESCAPE PRIVATE PLATFORM_MACOS)
elseif(UNIX)
    target_compile_definitions(GHOSTESCAPE PRIVATE PLATFORM_LINUX)
endif()

# 设置调试工作区目录
set_target_properties(GHOSTESCAPE PROPERTIES
    VS_DEBUG_WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
)

# 复制 assets 到可执行文件所在目录
add_custom_command(TARGET GHOSTESCAPE POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different
    ${PROJECT_SOURCE_DIR}/assets
    $<TARGET_FILE_DIR:GHOSTESCAPE>/assets
    COMMENT "Copying assets to build directory..."
)
